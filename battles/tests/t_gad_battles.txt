-- Writes only in its own folder (LIB_GAD/battles/tests/_work); never touches LIB_LLMIN.
local tDir, tW, tB1, tB2, tA, tR, tE, tS, tM, tN, tI, tOK, tFight, tSum1, tSum2, tWords, tSeed
local tU1, tU2, tU3, tXY, tCard, tTurn, tStack, tText, tF
put "/home/igame3d/projects/LIB_GAD/battles/" into tDir
put tDir & "tests/_work" into tW
if there is not a folder tW then create folder tW
start using stack (tDir & "gad_battles_rules.livecodescript")
put ck("library loads", the result, empty) & return after tLog

-- 1. the records
GADB.New 42
GADB.SetSetting "aiAttack", 0
put GADB.AddBase(1, 150, 450, 1, 50, "agent,militia") into tB1
put GADB.AddBase(2, 650, 140, 1, 34, "thrall") into tB2
put GADB.AddAnomaly(tB2, 3, 0, "thrall") into tA
put ck("ids b_001 b_002 a_001", tB1 && tB2 && tA, "b_001 b_002 a_001") & return after tLog
put ck("Base record has the plan's 14 items", the number of items of GADB.Record(tB1), 14) & return after tLog
put ck("anomaly Base grey (detect 0) at the start", GADB.Get(tB2, "detect"), 0) & return after tLog
put ck("anomaly knows its Base", GADB.Get(tA, "base") & "/" & GADB.Get(tB2, "anomaly"), "b_002/a_001") & return after tLog
GADB.SetSetting "travel", "fastest"
put ck("travel=slowest is the only mode", GADB.Setting("travel"), "slowest") & return after tLog

-- 2. a Base builds one unit at a time, toward its send-to (the rendezvous X)
put GADB.AddRendezvous(1, 300, 400) into tR
put ck("rendezvous r_001", tR, "r_001") & return after tLog
GADB.SetSendTo tB1, tR
get GADB.EndTurn()
put ck("turn 1: assembly 50, no unit yet", the number of lines of GADB.Ids("m"), 0) & return after tLog
get GADB.EndTurn()
put ck("turn 2: your Base (production 50) built one unit", the number of lines of GADB.Ids("m"), 1) & return after tLog
put ck("yours heads for the X", GADB.Get("m_001", "sendto"), tR) & return after tLog
get GADB.EndTurn()
put ck("turn 3: the anomaly's Base (production 34) built its first", GADB.Get("m_002", "owner"), 2) & return after tLog
put ck("theirs stays home (send-to = its Base)", GADB.Get("m_002", "sendto"), tB2) & return after tLog

-- 3. yours gather at the X into one squad (same team, same place, at the start of a turn)
put empty into tS
repeat 20
   get GADB.EndTurn()
   repeat for each line tM in GADB.Ids("s")
      if GADB.Get(tM, "owner") is 1 and GADB.XY(tM) is GADB.XY(tR) and GADB.Personnel(tM) >= 3 then put tM into tS
   end repeat
   if tS is not empty then exit repeat
end repeat
put "INFO after turn " & GADB.Turn() - 1 & ":" & return & GADB.State() & return after tLog
put ck("a squad of 3+ formed at the X", tS is not empty, true) & return after tLog
put ck("only one squad of yours at the X", countSquadsAt(1, GADB.XY(tR)), 1) & return after tLog
put ck("the squad has an agent and a militia", (GADB.Members(tS) contains "m_001") and (GADB.Members(tS) contains "m_003"), true) & return after tLog
put ck("squad moves at its slowest member's speed (militia 60)", GADB.Speed(tS), 60) & return after tLog
put ck("the computer's units became its garrison squad", the number of lines of GADB.Garrison(tB2), 1) & return after tLog

-- 4. send the squad to the anomaly's Base; the fight resolves itself on arrival
GADB.SetSendTo tB1, tB1
GADB.SetSendTo tS, tB2
put ck("8 turns to go (436 map units at 60)", GADB.Get(tS, "turnsLeft"), 8) & return after tLog
put empty into tFight
repeat 12
   put GADB.EndTurn() into tE
   if tE contains ("fight" & tab) then
      put line lineOffset("fight" & tab, tE) of tE into tFight
      exit repeat
   end if
end repeat
put "INFO fight: " & tFight & return after tLog
set the itemDelimiter to tab
put item 4 of tFight into tOK
set the itemDelimiter to comma
put ck("the fight happened on arrival", tFight is not empty, true) & return after tLog
put ck("result word TAKEN or HELD", tOK is among the items of "TAKEN,HELD", true) & return after tLog
put ck("anomaly Base was detected before the fight", GADB.Get(tB2, "detect") >= 1, true) & return after tLog
if tOK is "TAKEN" then
   put ck("TAKEN: the Base is yours now", GADB.Get(tB2, "owner"), 1) & return after tLog
   put ck("TAKEN: your squad guards it", GADB.Garrison(tB2) is not empty, true) & return after tLog
else
   put ck("HELD: the Base is still the computer's", GADB.Get(tB2, "owner"), 2) & return after tLog
   put ck("HELD: your squad is broken", GADB.Exists(tS), false) & return after tLog
end if

-- 5. deterministic: the same seed plays the same game; INFO how the seeds fall
put playLoop(42, 0) into tSum1
put playLoop(42, 0) into tSum2
put ck("same seed, same game", tSum1, tSum2) & return after tLog
put ck("the loop alone gives the same fight", item 2 of tSum1, tOK) & return after tLog
repeat for each item tI in "0,4"
   put empty into tWords
   repeat with tSeed = 1 to 20
      put item 2 of playLoop(tSeed, tI) & space after tWords
   end repeat
   put "INFO seeds 1..20, aiAttack " & tI & ": " & tWords & return after tLog
end repeat

-- 6. the fight handler by itself
GADB.New 5

repeat 4
   put GADB.AddUnit(1, "superhero", 380, 300, 1) into tM
end repeat

put ck("four at one place = one squad of 4", GADB.Personnel(tS), 4) & return after tLog
put ck("four superheroes take an empty level-1 Base", GADB.Fight(tS, tB2), "TAKEN") & return after tLog
put ck("the Base changed owner", GADB.Get(tB2, "owner"), 1) & return after tLog
put ck("taken Base fully known", GADB.Get(tB2, "detect"), 4) & return after tLog

   put GADB.AddUnit(2, "thrall", 400, 300, 3) into tM
   GADB.SetSendTo tM, tB2

-- the die tips an even fight: 3 agents against 3 agents, seeds 1..20
put 0 into tN
repeat with tSeed = 1 to 20
   GADB.New tSeed
   repeat 3
      get GADB.AddUnit(1, "agent", 200, 200, 1)
      get GADB.AddUnit(2, "agent", 260, 200, 1)
   end repeat
   GADB.FormSquads
   if GADB.Fight(line 1 of GADB.Ids("s"), line 2 of GADB.Ids("s")) is "WON" then add 1 to tN
end repeat
put "INFO even fights won by the attacker: " & tN & " of 20" & return after tLog
put ck("the seeded die decides even fights both ways", tN > 0 and tN < 20, true) & return after tLog
-- first strike: the same squads, the attacker faster (vigilantes 80 against militia 60, equal power x personnel)
put 0 into tN
repeat with tSeed = 1 to 20
   GADB.New tSeed
   GADB.SetKind "vigilante", 10, 80
   GADB.SetKind "militia", 10, 60
   repeat 3
      get GADB.AddUnit(1, "vigilante", 200, 200, 1)
      get GADB.AddUnit(2, "militia", 260, 200, 1)
   end repeat
   GADB.FormSquads
   if GADB.Fight(line 1 of GADB.Ids("s"), line 2 of GADB.Ids("s")) is "WON" then add 1 to tN
end repeat
put "INFO faster attacker won: " & tN & " of 20" & return after tLog
put ck("first strike: the faster side wins most even fights", tN >= 15, true) & return after tLog

--HANDLERS--
function countSquadsAt pOwner, pXY
   local tN, tM
   put 0 into tN
   repeat for each line tM in GADB.Movers()
      if GADB.Get(tM, "owner") is pOwner and GADB.XY(tM) is pXY then add 1 to tN
   end repeat
   return tN
end countSquadsAt

-- the whole loop from a seed (pAI: the computer's aiAttack): "turn of the fight,word,owner of b_002 after"
function playLoop pSeed, pAI
   local tB1, tB2, tR, tS, tM, tE, tI
   GADB.New pSeed
   GADB.SetSetting "aiAttack", pAI
   put GADB.AddBase(1, 150, 450, 1, 50, "agent,militia") into tB1
   put GADB.AddBase(2, 650, 140, 1, 34, "thrall") into tB2
   get GADB.AddAnomaly(tB2, 3, 0, "thrall")
   put GADB.AddRendezvous(1, 300, 400) into tR
   GADB.SetSendTo tB1, tR
   repeat 22
      get GADB.EndTurn()
      put empty into tS
      repeat for each line tM in GADB.Ids("s")
         if GADB.Get(tM, "owner") is 1 and GADB.XY(tM) is GADB.XY(tR) and GADB.Personnel(tM) >= 3 then put tM into tS
      end repeat
      if tS is not empty then exit repeat
   end repeat
   if tS is empty then return "no squad"
   GADB.SetSendTo tB1, tB1
   GADB.SetSendTo tS, tB2
   repeat 12
      put GADB.EndTurn() into tE
      if tE contains ("fight" & tab) then
         set the itemDelimiter to tab
         put item 4 of line lineOffset("fight" & tab, tE) of tE into tE
         set the itemDelimiter to comma
         return GADB.Turn() - 1 & "," & tE & "," & GADB.Get(tB2, "owner")
      end if
   end repeat
   return "no fight"
end playLoop
